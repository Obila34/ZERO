# ZERO — Pi-side VISION extras (the eyes). Install on the Pi in addition to
# requirements.txt, only if you enable vision (vision.enabled: true):
#
#   pip install -r requirements-vision.txt
#
# Deliberately LIGHT: YOLO11n runs through ONNX Runtime (the same runtime the
# wake word already uses), so the Pi needs NO torch / CUDA / ultralytics — those
# drag ~2 GB of NVIDIA wheels onto ARM and fill the SD card. The model ships as
# the checked-in yolo11n.onnx (regenerate with scripts/export_yolo_onnx.py).
#
# The GPU vision server has its own deps (torch + transformers) in
# server/vision/requirements.txt — install those on the GPU node, not the Pi.

opencv-python>=4.8        # camera capture, color (HSV), JPEG encode
onnxruntime>=1.16         # YOLO11n inference (usually already present for the wake word)
pydantic>=2.0             # the vision wire schema (Detection / SceneFact / ...)
